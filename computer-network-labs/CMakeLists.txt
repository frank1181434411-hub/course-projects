cmake_minimum_required(VERSION 3.25)
project(ComputerNetworkLabs LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Windows with MSVC and Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(net_common INTERFACE)
target_include_directories(net_common INTERFACE common)
target_link_libraries(net_common INTERFACE Ws2_32)
foreach(lab IN ITEMS lab02 lab03)
  if(lab STREQUAL "lab02")
    set(src lab02-winsock-basics/src)
  else()
    set(src lab03-windows-socket/src)
  endif()
  foreach(role IN ITEMS server client)
    add_executable(${lab}_${role} ${src}/${role}.cpp)
    target_link_libraries(${lab}_${role} PRIVATE net_common)
  endforeach()
endforeach()
add_executable(lab03_ftp WIN32 lab03-windows-socket/src/ftp_gui.cpp)
target_link_libraries(lab03_ftp PRIVATE net_common User32 Gdi32)
